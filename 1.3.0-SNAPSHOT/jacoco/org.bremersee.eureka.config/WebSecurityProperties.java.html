package org.bremersee.eureka.config;

import lombok.Data;
import org.springframework.boot.context.properties.ConfigurationProperties;

/**
 * The web security properties.
 *
 * @author Christian Bremer
 */
@ConfigurationProperties(prefix = "bremersee.eureka.authentication")
@Data
public class WebSecurityProperties {

  /**
   * Required role to access application endpoints.
   */
  public static final String ROLE_APPLICATION = "ROLE_EUREKA_SERVER_ADMIN";

  /**
   * The required role to access the config server.
   */
  private String eurekaServerAccessRole = ROLE_APPLICATION;

  /**
   * Instantiates new web security properties.
   */
  public WebSecurityProperties() {
    super();
  }
}
